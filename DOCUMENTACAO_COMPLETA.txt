import os
from datetime import datetime

# ════════════════════════════════════════════════════════════════
# CONFIGURAÇÕES
# ════════════════════════════════════════════════════════════════

CONFIG = {
    "pasta_raiz": os.path.dirname(os.path.abspath(__file__)),
    "arquivo_saida": "DOCUMENTACAO_COMPLETA.txt",
    "extensoes_permitidas": [".py", ".html", ".css", ".txt", ".h", ".c", ".md", ".cfg", ".toml"],
    "ignorar": [
        "venv",
        ".venv",
        "__pycache__",
        ".git",
        "staticfiles",
        "db.sqlite3",
        "DOCUMENTACAO_COMPLETA.txt",
        "gerar_documentacao.py",
        ".env",
        "migrations",
    ],
    "tamanho_maximo": 500_000,  # 500 KB
}


def deve_ignorar(caminho):
    nome = os.path.basename(caminho)
    return any(item in caminho or nome == item for item in CONFIG["ignorar"])


def deve_incluir(arquivo):
    _, ext = os.path.splitext(arquivo)
    return ext in CONFIG["extensoes_permitidas"]


def obter_estrutura_pasta(diretorio, nivel=0):
    estrutura = ""
    indent = "  " * nivel

    try:
        itens = sorted(os.listdir(diretorio))
    except Exception as erro:
        return f"Erro ao ler {diretorio}: {erro}\n"

    for item in itens:
        caminho_completo = os.path.join(diretorio, item)

        if deve_ignorar(caminho_completo):
            continue

        if os.path.isdir(caminho_completo):
            estrutura += f"{indent}[PASTA] {item}/\n"
            estrutura += obter_estrutura_pasta(caminho_completo, nivel + 1)
        elif deve_incluir(item):
            tamanho = os.path.getsize(caminho_completo) / 1024
            estrutura += f"{indent}[ARQ] {item} ({tamanho:.2f} KB)\n"

    return estrutura


def obter_todos_arquivos(diretorio, lista_arquivos=None):
    if lista_arquivos is None:
        lista_arquivos = []

    try:
        itens = os.listdir(diretorio)
    except Exception as erro:
        print(f"Erro ao processar {diretorio}: {erro}")
        return lista_arquivos

    for item in itens:
        caminho_completo = os.path.join(diretorio, item)

        if deve_ignorar(caminho_completo):
            continue

        if os.path.isdir(caminho_completo):
            obter_todos_arquivos(caminho_completo, lista_arquivos)
        elif deve_incluir(item):
            tamanho = os.path.getsize(caminho_completo)
            lista_arquivos.append({
                "caminho": caminho_completo,
                "caminho_relativo": os.path.relpath(caminho_completo, CONFIG["pasta_raiz"]),
                "tamanho": tamanho,
                "nome": item,
            })

    return lista_arquivos


def gerar_cabecalho():
    data = datetime.now().strftime("%d/%m/%Y %H:%M:%S")
    return f"""
{"=" * 79}
                    DOCUMENTACAO COMPLETA DO PROJETO
{"=" * 79}

Gerado em: {data}
Diretorio: {CONFIG["pasta_raiz"]}

{"=" * 79}

"""


def gerar_indice(arquivos):
    indice = f"""
{"=" * 79}
                          INDICE DE ARQUIVOS
{"=" * 79}

Total de arquivos: {len(arquivos)}

"""
    for i, arquivo in enumerate(arquivos):
        tamanho = arquivo["tamanho"] / 1024
        indice += f"{str(i + 1).zfill(3)}. {arquivo['caminho_relativo']} ({tamanho:.2f} KB)\n"

    indice += f"\n{'=' * 79}\n\n"
    return indice


def gerar_estrutura():
    estrutura = f"""
{"=" * 79}
                      ESTRUTURA DE PASTAS
{"=" * 79}

"""
    estrutura += obter_estrutura_pasta(CONFIG["pasta_raiz"])
    estrutura += f"\n{'=' * 79}\n\n"
    return estrutura


def gerar_conteudo_arquivo(arquivo):
    mod_time = datetime.fromtimestamp(
        os.path.getmtime(arquivo["caminho"])
    ).strftime("%d/%m/%Y %H:%M:%S")

    conteudo = f"""
{"=" * 79}
ARQUIVO: {arquivo["caminho_relativo"]}
{"=" * 79}

Caminho completo: {arquivo["caminho"]}
Tamanho: {arquivo["tamanho"] / 1024:.2f} KB
Ultima modificacao: {mod_time}

{"-" * 79}

"""

    try:
        with open(arquivo["caminho"], "r", encoding="utf-8", errors="replace") as f:
            codigo = f.read()

        if arquivo["tamanho"] > CONFIG["tamanho_maximo"]:
            linhas = codigo.splitlines()
            codigo = "\n".join(linhas[:1000])
            codigo += "\n\n[ARQUIVO TRUNCADO: mostrando apenas as primeiras 1000 linhas]\n"

        conteudo += codigo

    except Exception as erro:
        conteudo += f"ERRO AO LER ARQUIVO: {erro}\n"

    conteudo += f"\n\n{'=' * 79}\n"
    return conteudo


def gerar_estatisticas(arquivos):
    por_extensao = {}
    tamanho_total = 0

    for arquivo in arquivos:
        _, extensao = os.path.splitext(arquivo["nome"])
        por_extensao[extensao] = por_extensao.get(extensao, 0) + 1
        tamanho_total += arquivo["tamanho"]

    estatisticas = f"""
{"=" * 79}
                          ESTATISTICAS
{"=" * 79}

Total de arquivos: {len(arquivos)}
Tamanho total: {tamanho_total / 1024 / 1024:.2f} MB

Arquivos por tipo:
"""

    for extensao, quantidade in sorted(
        por_extensao.items(),
        key=lambda item: item[1],
        reverse=True,
    ):
        estatisticas += f"   {extensao or '[sem extensao]'}: {quantidade} arquivo(s)\n"

    estatisticas += f"\n{'=' * 79}\n\n"
    return estatisticas


def executar():
    print("Iniciando a geracao da documentacao...\n")
    inicio = datetime.now()

    print("Escaneando arquivos...")
    arquivos = obter_todos_arquivos(CONFIG["pasta_raiz"])
    arquivos.sort(key=lambda arquivo: arquivo["caminho_relativo"].lower())
    print(f"{len(arquivos)} arquivos encontrados.\n")

    documentacao = ""
    documentacao += gerar_cabecalho()
    documentacao += gerar_estatisticas(arquivos)
    documentacao += gerar_estrutura()
    documentacao += gerar_indice(arquivos)

    print("Adicionando conteudo dos arquivos...")
    for numero, arquivo in enumerate(arquivos, start=1):
        print(f"[{numero}/{len(arquivos)}] {arquivo['caminho_relativo']}")
        documentacao += gerar_conteudo_arquivo(arquivo)

    caminho_saida = os.path.join(
        CONFIG["pasta_raiz"],
        CONFIG["arquivo_saida"],
    )

    with open(caminho_saida, "w", encoding="utf-8") as f:
        f.write(documentacao)

    duracao = (datetime.now() - inicio).total_seconds()
    tamanho_saida = os.path.getsize(caminho_saida)

    print("\nConcluido!")
    print(f"Arquivo gerado: {caminho_saida}")
    print(f"Tamanho: {tamanho_saida / 1024 / 1024:.2f} MB")
    print(f"Tempo: {duracao:.2f} segundos")


if __name__ == "__main__":
    try:
        executar()
    except Exception as erro:
        print(f"Erro fatal: {erro}")
        raise

# Beyond the Fallen

**Beyond the Fallen** é um jogo de ação e combate desenvolvido em C como projeto da matéria de Programação Imperativa e Funcional (PIF), da CESAR School. A aventura se passa em um gulag cercado pela neve, onde um prisioneiro precisa sobreviver aos confrontos, aprender habilidades e encontrar uma saída.

> **Status:** conceito e desenvolvimento em andamento. Os detalhes e as mecânicas descritos aqui representam a proposta do jogo e podem mudar durante a implementação.

## História

Você é um prisioneiro soviético em um gulag. Para conquistar a liberdade, terá de enfrentar outros detentos — cada um com sua própria história, personalidade e estilo de luta.

Enquanto avança, você poderá aprender novas habilidades e encontrar armas brancas para aumentar suas chances de sobreviver. Mas o gulag guarda mistérios além das arenas: perto do portão, é possível ouvir batidas e gritos, enquanto homens de jaleco entram e saem de uma porta carregando seringas.

O que acontece ali? E o que o último desafio tem a ver com isso?

## Progressão e chefes

A campanha apresenta confrontos contra chefes com características e padrões de combate próprios. Entre os adversários planejados estão:

- **Bobrovsky** — um russo corpulento, preso por trair a pátria. Sua força e seu tamanho fazem dele um adversário difícil de derrubar.
- **Vladimir** — um lituano ágil e habilidoso. Apesar de rápido, o medo pode atrapalhar suas decisões.
- **Ivan** — um bielorrusso muito forte, mas lento. Seus golpes são perigosos, porém podem ser evitados com atenção.
- **Anastasia** — o desafio final. O nome pode sugerir uma criança de oito anos, mas...

Ao vencer os desafios, o jogador poderá desenvolver novas habilidades e usar armas encontradas durante a jornada.

## Mundo e cenários

O gulag funciona como um mapa fechado para explorar. Entre os confrontos, o jogador poderá circular por áreas como:

- A cela;
- O refeitório;
- A área central;
- O corredor de entrada e saída.

O corredor próximo ao portão abriga um mistério que se revela aos poucos por meio de sons e da movimentação dos homens de jaleco.

A direção visual proposta é simples e fria, com neve, pedra e estruturas austeras. A batalha final terá uma arena de pedra marcada por tremores, reforçando a força do confronto.

## Combate

O sistema de combate foi pensado para ser direto e acessível, com opções como:

- Atacar;
- Esquivar;
- Defender;
- Pular;
- Travar a mira no adversário;
- Usar armas brancas e itens.

Cada chefe exige atenção aos seus movimentos e ao momento certo para atacar ou se defender.

## Destaques

- Ambientação inspirada em um gulag da URSS;
- Chefes com histórias, personalidades e estilos de luta diferentes;
- Progressão com novas habilidades e armas;
- Exploração de um mapa fechado entre os combates;
- Mistérios espalhados pelo gulag;
- Diálogos com humor e uma surpresa no confronto final.

## Tecnologias

- **Linguagem:** C

## Desenvolvimento

Este projeto está em desenvolvimento. A quantidade de chefes, as habilidades, os itens, os controles e os detalhes da história podem ser ajustados ao longo da criação do jogo.

## Autoria

Projeto desenvolvido para a matéria de PIF da CESAR School.

#include "combat.h"
#include "config.h"

static int calcular_dano(Player *p) {
    if (p->dano < 0) {
        return 0;
    }

    return p->dano;
}

void combat_atacar(Player *p, Boss *b) {
    if (p->hp <= 0 || b->hp <= 0) {
        return;
    }

    b->hp -= calcular_dano(p);

    if (b->hp < 0) {
        b->hp = 0;
    }
}

void combat_defender(Player *p) {
    if (p->hp <= 0 || p->cooldown_defesa > 0) {
        return;
    }

    p->defendendo = 1;
    p->esquivando = 0;
}

void combat_esquivar(Player *p) {
    if (p->hp <= 0 || p->cooldown_esquiva > 0) {
        return;
    }

    p->esquivando = 1;
    p->defendendo = 0;
}

void combat_pular(Player *p) {
    combat_esquivar(p);
}

void combat_boss_atacar(Boss *b, Player *p) {
    if (p->hp <= 0 || b->hp <= 0) {
        return;
    }

    int dano = b->dano;

    if (p->esquivando) {
        p->esquivando = 0;
        p->cooldown_esquiva = COOLDOWN_ESQUIVA;
        return;
    }

    if (p->defendendo) {
        dano = (dano + 1) / 2;

        p->defendendo = 0;
        p->cooldown_defesa = COOLDOWN_DEFESA;
    }

    p->hp -= dano;

    if (p->hp < 0) {
        p->hp = 0;
    }
}

void combat_executar_acao(Player *p, Boss *b, Acao acao) {
    if (p->hp <= 0 || b->hp <= 0) {
        return;
    }

    switch (acao) {
        case ACAO_ATACAR:
            combat_atacar(p, b);
            break;

        case ACAO_DEFENDER:
            combat_defender(p);
            break;

        case ACAO_ESQUIVAR:
            combat_esquivar(p);
            break;

        case ACAO_PULAR:
            combat_pular(p);
            break;
    }
}

void combat_atualizar(Player *p, float tempo) {
    if (tempo < 0) {
        return;
    }

    if (p->cooldown_defesa > 0) {
        p->cooldown_defesa -= tempo;

        if (p->cooldown_defesa < 0) {
            p->cooldown_defesa = 0;
        }
    }

    if (p->cooldown_esquiva > 0) {
        p->cooldown_esquiva -= tempo;

        if (p->cooldown_esquiva < 0) {
            p->cooldown_esquiva = 0;
        }
    }
}

int combat_verificar_fim(Player *p, Boss *b) {
    if (p->hp <= 0 && b->hp <= 0) {
        return COMBATE_EMPATE;
    }

    if (b->hp <= 0) {
        return COMBATE_VITORIA;
    }

    if (p->hp <= 0) {
        return COMBATE_DERROTA;
    }

    return COMBATE_EM_ANDAMENTO;
}

#include "raylib.h"
#include "player.h"
#include "boss.h"
#include "combat.h"
#include "config.h"

typedef enum {
    TELA_COMBATE,
    TELA_FINAL
} Tela;

typedef enum {
    BOSS_GUARDANDO,
    BOSS_PREPARANDO,
    BOSS_RECUPERANDO
} EstadoBoss;

typedef enum {
    ATAQUE_ESQUERDA,
    ATAQUE_DIREITA
} DirecaoAtaque;

typedef enum {
    ESQUIVA_NENHUMA,
    ESQUIVA_ESQUERDA,
    ESQUIVA_DIREITA
} DirecaoEsquiva;

static void iniciar_boss(Boss *boss, int numero_boss) {
    switch (numero_boss) {
        case 1:
            boss_iniciar_bobrovsky(boss);
            break;
        case 2:
            boss_iniciar_vladimir(boss);
            break;
        case 3:
            boss_iniciar_ivan(boss);
            break;
        case 4:
            boss_iniciar_anastasia(boss);
            break;
        default:
            boss_iniciar_bobrovsky(boss);
            break;
    }
}

static int obter_vida_maxima_boss(int numero_boss) {
    switch (numero_boss) {
        case 1: return BOBROVSKY_VIDA;
        case 2: return VLADIMIR_VIDA;
        case 3: return IVAN_VIDA;
        case 4: return ANASTASIA_VIDA;
        default: return BOBROVSKY_VIDA;
    }
}

/*
 * Desenha o jogador em primeiro plano, visto por trás.
 * O deslocamento horizontal representa a esquiva.
 */
static void desenhar_jogador_de_costas(
    int centro_x,
    int topo_y,
    int atacando,
    int defendendo,
    int esquivando,
    int direcao_esquiva
) {
    int deslocamento_x = 0;
    int deslocamento_y = 0;

    if (esquivando) {
        deslocamento_x = direcao_esquiva * 35;
        deslocamento_y = 10;
    }

    int x = centro_x + deslocamento_x;
    int y = topo_y + deslocamento_y;

    Color roupa = (Color){25, 75, 145, 255};
    Color roupa_sombra = (Color){17, 52, 105, 255};
    Color pele = (Color){218, 178, 132, 255};
    Color luva = (Color){190, 35, 35, 255};
    Color calca = (Color){30, 40, 58, 255};

    // Pernas e botas.
    DrawRectangle(x - 52, y + 185, 38, 75, calca);
    DrawRectangle(x + 14, y + 185, 38, 75, calca);
    DrawRectangle(x - 60, y + 245, 48, 19, (Color){35, 31, 29, 255});
    DrawRectangle(x + 12, y + 245, 48, 19, (Color){35, 31, 29, 255});

    // Tronco largo para reforçar a perspectiva de câmera atrás do jogador.
    DrawEllipse(x, y + 137, 76, 83, roupa);
    DrawRectangle(x - 63, y + 98, 126, 75, roupa);

    // Costura/detalhe nas costas.
    DrawRectangle(x - 6, y + 105, 12, 92, roupa_sombra);
    DrawRectangle(x - 43, y + 139, 86, 8, roupa_sombra);

    // Cabeça vista por trás.
    DrawCircle(x, y + 58, 39, pele);
    DrawRectangle(x - 39, y + 42, 78, 27, (Color){65, 43, 31, 255});
    DrawRectangle(x - 31, y + 25, 62, 20, (Color){65, 43, 31, 255});

    if (defendendo) {
        // Guarda alta: braços levantados na frente do corpo.
        DrawLineEx(
            (Vector2){(float)(x - 53), (float)(y + 115)},
            (Vector2){(float)(x - 35), (float)(y + 64)},
            22,
            roupa
        );
        DrawLineEx(
            (Vector2){(float)(x + 53), (float)(y + 115)},
            (Vector2){(float)(x + 35), (float)(y + 64)},
            22,
            roupa
        );
        DrawCircle(x - 35, y + 61, 17, luva);
        DrawCircle(x + 35, y + 61, 17, luva);
    } else if (atacando) {
        // Soco direcionado ao adversário, acima da silhueta do ombro.
        DrawLineEx(
            (Vector2){(float)(x + 25), (float)(y + 122)},
            (Vector2){(float)(x + 8), (float)(y + 38)},
            23,
            roupa
        );
        DrawLineEx(
            (Vector2){(float)(x - 24), (float)(y + 123)},
            (Vector2){(float)(x - 53), (float)(y + 84)},
            21,
            roupa
        );
        DrawCircle(x + 8, y + 35, 18, luva);
        DrawCircle(x - 57, y + 80, 16, luva);
    } else {
        // Guarda neutra, com luvas visíveis nas laterais.
        DrawLineEx(
            (Vector2){(float)(x - 53), (float)(y + 115)},
            (Vector2){(float)(x - 70), (float)(y + 165)},
            21,
            roupa
        );
        DrawLineEx(
            (Vector2){(float)(x + 53), (float)(y + 115)},
            (Vector2){(float)(x + 70), (float)(y + 165)},
            21,
            roupa
        );
        DrawCircle(x - 72, y + 169, 17, luva);
        DrawCircle(x + 72, y + 169, 17, luva);
    }
}

static void reiniciar_combate(
    Player *jogador,
    Boss *boss,
    int numero_boss,
    EstadoBoss *estado_boss,
    float *tempo_estado,
    float *tempo_soco,
    float *tempo_recuperacao_soco,
    float *tempo_contra_ataque,
    int *quantidade_ataques,
    int *soco_esquerdo,
    int *soco_direito,
    int *esquiva_perfeita,
    int *resultado
) {
    iniciar_boss(boss, numero_boss);

    jogador->hp = PLAYER_VIDA;
    jogador->dano = PLAYER_DANO_BASE;
    jogador->defendendo = 0;
    jogador->esquivando = 0;
    jogador->cooldown_defesa = 0;
    jogador->cooldown_esquiva = 0;

    *estado_boss = BOSS_GUARDANDO;
    *tempo_estado = 0;
    *tempo_soco = 0;
    *tempo_recuperacao_soco = 0;
    *tempo_contra_ataque = 0;
    *quantidade_ataques = 0;
    *soco_esquerdo = 0;
    *soco_direito = 0;
    *esquiva_perfeita = 0;
    *resultado = COMBATE_EM_ANDAMENTO;
}

int main(void) {
    Player jogador;
    Boss boss;

    player_iniciar(&jogador);

    InitWindow(LARGURA_TELA, ALTURA_TELA, "Beyond the Fallen");
    SetTargetFPS(60);

    Tela tela = TELA_COMBATE;
    EstadoBoss estado_boss = BOSS_GUARDANDO;

    DirecaoAtaque direcao_ataque = ATAQUE_ESQUERDA;
    DirecaoEsquiva direcao_esquiva = ESQUIVA_NENHUMA;

    int numero_boss = 1;
    int resultado = COMBATE_EM_ANDAMENTO;
    int quantidade_ataques = 0;
    int esquiva_perfeita = 0;

    float tempo_estado = 0;
    float tempo_soco = 0;
    float tempo_recuperacao_soco = 0;
    float tempo_contra_ataque = 0;
    float intervalo_ataque = 2.0f;

    const float duracao_preparacao = 0.9f;
    const float duracao_recuperacao = 1.5f;

    int soco_esquerdo = 0;
    int soco_direito = 0;

    reiniciar_combate(
        &jogador,
        &boss,
        numero_boss,
        &estado_boss,
        &tempo_estado,
        &tempo_soco,
        &tempo_recuperacao_soco,
        &tempo_contra_ataque,
        &quantidade_ataques,
        &soco_esquerdo,
        &soco_direito,
        &esquiva_perfeita,
        &resultado
    );

    intervalo_ataque = 2.8f - boss.velocidade * 0.15f;
    if (intervalo_ataque < 1.4f) {
        intervalo_ataque = 1.4f;
    }

    while (!WindowShouldClose()) {
        float tempo = GetFrameTime();

        if (tela == TELA_COMBATE) {
            combat_atualizar(&jogador, tempo);

            if (tempo_recuperacao_soco > 0) {
                tempo_recuperacao_soco -= tempo;
                if (tempo_recuperacao_soco < 0) {
                    tempo_recuperacao_soco = 0;
                }
            }

            if (tempo_contra_ataque > 0) {
                tempo_contra_ataque -= tempo;
                if (tempo_contra_ataque < 0) {
                    tempo_contra_ataque = 0;
                }
            }

            if (tempo_soco > 0) {
                tempo_soco -= tempo;
                if (tempo_soco <= 0) {
                    tempo_soco = 0;
                    soco_esquerdo = 0;
                    soco_direito = 0;
                }
            }

            // Socos acertam quando o chefe está se recuperando.
            if ((IsKeyPressed(KEY_J) || IsKeyPressed(KEY_K)) &&
                tempo_recuperacao_soco <= 0 &&
                jogador.hp > 0 &&
                boss.hp > 0 &&
                estado_boss == BOSS_RECUPERANDO) {

                soco_esquerdo = IsKeyPressed(KEY_J);
                soco_direito = IsKeyPressed(KEY_K);
                tempo_soco = 0.2f;
                tempo_recuperacao_soco = 0.35f;

                int dano_original = jogador.dano;

                if (tempo_contra_ataque > 0 && esquiva_perfeita) {
                    jogador.dano = PLAYER_DANO_BASE * 2;
                    tempo_contra_ataque = 0;
                    esquiva_perfeita = 0;
                }

                combat_atacar(&jogador, &boss);
                jogador.dano = dano_original;
            }

            if (IsKeyPressed(KEY_S)) {
                combat_defender(&jogador);
            }

            if (IsKeyPressed(KEY_A) &&
                jogador.cooldown_esquiva <= 0) {
                combat_esquivar(&jogador);
                direcao_esquiva = ESQUIVA_ESQUERDA;
            }

            if (IsKeyPressed(KEY_D) &&
                jogador.cooldown_esquiva <= 0) {
                combat_esquivar(&jogador);
                direcao_esquiva = ESQUIVA_DIREITA;
            }

            tempo_estado += tempo;

            switch (estado_boss) {
                case BOSS_GUARDANDO:
                    if (tempo_estado >= intervalo_ataque) {
                        estado_boss = BOSS_PREPARANDO;
                        tempo_estado = 0;
                        direcao_esquiva = ESQUIVA_NENHUMA;
                        direcao_ataque =
                            GetRandomValue(0, 1) == 0
                                ? ATAQUE_ESQUERDA
                                : ATAQUE_DIREITA;
                    }
                    break;

                case BOSS_PREPARANDO:
                    if (tempo_estado >= duracao_preparacao) {
                        quantidade_ataques++;

                        int esquivou_corretamente =
                            (direcao_ataque == ATAQUE_ESQUERDA &&
                             direcao_esquiva == ESQUIVA_DIREITA) ||
                            (direcao_ataque == ATAQUE_DIREITA &&
                             direcao_esquiva == ESQUIVA_ESQUERDA);

                        if (esquivou_corretamente) {
                            jogador.esquivando = 0;
                            jogador.cooldown_esquiva = COOLDOWN_ESQUIVA;
                            esquiva_perfeita = 1;
                            tempo_contra_ataque = 0.8f;
                        } else {
                            jogador.esquivando = 0;

                            int dano_original = boss.dano;

                            if (numero_boss == 4 &&
                                quantidade_ataques % 3 == 0) {
                                boss.dano = ANASTASIA_DANO_ESPECIAL;
                            }

                            combat_boss_atacar(&boss, &jogador);
                            boss.dano = dano_original;

                            esquiva_perfeita = 0;
                            tempo_contra_ataque = 0;
                        }

                        direcao_esquiva = ESQUIVA_NENHUMA;
                        estado_boss = BOSS_RECUPERANDO;
                        tempo_estado = 0;
                    }
                    break;

                case BOSS_RECUPERANDO:
                    if (tempo_estado >= duracao_recuperacao) {
                        estado_boss = BOSS_GUARDANDO;
                        tempo_estado = 0;
                        esquiva_perfeita = 0;
                        tempo_contra_ataque = 0;
                    }
                    break;
            }

            resultado = combat_verificar_fim(&jogador, &boss);
            if (resultado != COMBATE_EM_ANDAMENTO) {
                tela = TELA_FINAL;
            }
        } else if (tela == TELA_FINAL) {
            if (resultado == COMBATE_VITORIA &&
                numero_boss < 4 &&
                IsKeyPressed(KEY_ENTER)) {

                numero_boss++;

                reiniciar_combate(
                    &jogador,
                    &boss,
                    numero_boss,
                    &estado_boss,
                    &tempo_estado,
                    &tempo_soco,
                    &tempo_recuperacao_soco,
                    &tempo_contra_ataque,
                    &quantidade_ataques,
                    &soco_esquerdo,
                    &soco_direito,
                    &esquiva_perfeita,
                    &resultado
                );

                direcao_esquiva = ESQUIVA_NENHUMA;

                intervalo_ataque = 2.8f - boss.velocidade * 0.15f;
                if (intervalo_ataque < 1.4f) {
                    intervalo_ataque = 1.4f;
                }

                tela = TELA_COMBATE;
            } else if (resultado == COMBATE_DERROTA &&
                       IsKeyPressed(KEY_R)) {

                reiniciar_combate(
                    &jogador,
                    &boss,
                    numero_boss,
                    &estado_boss,
                    &tempo_estado,
                    &tempo_soco,
                    &tempo_recuperacao_soco,
                    &tempo_contra_ataque,
                    &quantidade_ataques,
                    &soco_esquerdo,
                    &soco_direito,
                    &esquiva_perfeita,
                    &resultado
                );

                direcao_esquiva = ESQUIVA_NENHUMA;
                tela = TELA_COMBATE;
            }
        }

        BeginDrawing();

        if (tela == TELA_COMBATE) {
            desenhar_arena();

            DrawText("BEYOND THE FALLEN", 326, 28, 28, RAYWHITE);

            // Vida do jogador.
            DrawText(jogador.nome, 55, 77, 22, YELLOW);
            DrawText(TextFormat("%d", jogador.hp), 55, 105, 19, RAYWHITE);
            DrawRectangle(95, 111, 260, 17, DARKGRAY);

            int hp_jogador = jogador.hp;
            if (hp_jogador < 0) hp_jogador = 0;
            if (hp_jogador > PLAYER_VIDA) hp_jogador = PLAYER_VIDA;

            DrawRectangle(
                95, 111,
                (int)(260.0f * hp_jogador / PLAYER_VIDA),
                17,
                (Color){35, 210, 75, 255}
            );

            // Vida do chefe.
            int hp_maximo_boss = obter_vida_maxima_boss(numero_boss);
            int hp_boss = boss.hp;

            if (hp_boss < 0) hp_boss = 0;
            if (hp_boss > hp_maximo_boss) hp_boss = hp_maximo_boss;

            DrawText(boss.nome, 650, 77, 22, RED);
            DrawText(TextFormat("%d", boss.hp), 650, 105, 19, RAYWHITE);
            DrawRectangle(690, 111, 220, 17, DARKGRAY);
            DrawRectangle(
                690, 111,
                (int)(220.0f * hp_boss / hp_maximo_boss),
                17,
                (Color){220, 45, 55, 255}
            );

            Color roupa_boss = (Color){155, 45, 45, 255};

            if (estado_boss == BOSS_PREPARANDO) {
                roupa_boss = (Color){205, 85, 40, 255};
            } else if (estado_boss == BOSS_RECUPERANDO) {
                roupa_boss = (Color){115, 55, 55, 255};
            }

            // Sombras no piso para dar profundidade.
            DrawEllipse(480, 386, 37, 9, (Color){15, 16, 18, 120});
            DrawEllipse(480, 505, 92, 15, (Color){15, 16, 18, 150});

            // Chefe ao fundo, de frente para o jogador.
            desenhar_boss_de_frente(
                480,
                157,
                roupa_boss,
                (Color){218, 178, 132, 255},
                estado_boss == BOSS_PREPARANDO,
                estado_boss == BOSS_PREPARANDO
            );

            // Jogador em primeiro plano, visto de costas.
            int direcao_visual_esquiva = 0;
            if (direcao_esquiva == ESQUIVA_ESQUERDA) {
                direcao_visual_esquiva = -1;
            } else if (direcao_esquiva == ESQUIVA_DIREITA) {
                direcao_visual_esquiva = 1;
            }

            desenhar_jogador_de_costas(
                480,
                270,
                tempo_soco > 0 && (soco_esquerdo || soco_direito),
                jogador.defendendo,
                jogador.esquivando,
                direcao_visual_esquiva
            );

            if (jogador.defendendo) {
                DrawEllipseLines(
                    480, 386, 91, 113,
                    (Color){80, 170, 255, 180}
                );
            }

            // Controles.
            DrawText("J/K: SOCO", 95, 475, 18, RAYWHITE);
            DrawText("S: DEFESA", 400, 475, 18, RAYWHITE);
            DrawText("A/D: ESQUIVA", 700, 475, 18, RAYWHITE);

            DrawText(
                TextFormat("ESQUIVA %.1f", jogador.cooldown_esquiva),
                95, 505, 16,
                (Color){190, 195, 205, 255}
            );
            DrawText(
                TextFormat("DEFESA %.1f", jogador.cooldown_defesa),
                400, 505, 16,
                (Color){190, 195, 205, 255}
            );
        } else if (tela == TELA_FINAL) {
            ClearBackground((Color){22, 24, 29, 255});

            if (resultado == COMBATE_VITORIA && numero_boss < 4) {
                DrawText("VITORIA", 365, 170, 52, GREEN);
                DrawText(
                    "ENTER: PRÓXIMO COMBATE",
                    285, 275, 22, RAYWHITE
                );
            } else if (resultado == COMBATE_VITORIA) {
                DrawText(
                    "CAMPEÃO",
                    360, 170, 52,
                    (Color){255, 205, 55, 255}
                );
                DrawText(
                    "VOCÊ VENCEU A CAMPANHA",
                    280, 255, 24, RAYWHITE
                );

                );
            } else {

    }

}
